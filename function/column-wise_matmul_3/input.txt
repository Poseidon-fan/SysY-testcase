const int N = 500;

int state;

int get_random() {
    int n = state % 1024;
    int i = 0;
    while (i < n) {
        state = state + 128;
        state = state % 65535;
        i = i + 1;
    }
    state = state % 65535;
    return state;
}

void init_matrix(int m[][N]) {
    int i = 0;
    while (i < N) {
        int j = 0;
        while (j < N) {
            m[i][j] = get_random();
            j = j + 1;
        }
        i = i + 1;
    }
}

void print_matrix(int m[][N]) {
    int i = 0;
    while (i < N) {
        int j = 0;
        while (j < N) {
            putint(m[i][j]);
            putch(32);
            j = j + 1;
        }
        putch(10);
        i = i + 1;
    }
}

void transpose_matrix(int origin[][N], int transposed[][N]) { // column-wise access
    int i = 0;
    while (i < N) {
        int j = 0;
        while (j < N) {
            transposed[j][i] = origin[i][j];
            j = j + 1;
        }
        i = i + 1;
    }
}

void multiply_matrix(int a[][N], int b[][N], int result[][N]) { // column-wise access
    int i = 0;
    while (i < N) {
        int j = 0;
        while (j < N) {
            int sum = 0;
            int k = 0;
            while (k < N) {
                sum = sum + a[i][k] * b[k][j];
                k = k + 1;
            }
            result[i][j] = sum;
            j = j + 1;
        }
        i = i + 1;
    }
}

int main() {
    int A[N][N], B[N][N], C[N][N], B_transposed[N][N];
    
    state = getint();
    int N = getint();

    starttime();

    init_matrix(A);
    init_matrix(B);
    init_matrix(C);

    int k = 0;
    while (k < N) {
        transpose_matrix(B, B_transposed);
        multiply_matrix(A, B_transposed, C);
        k = k + 1;
    }

    stoptime();

    print_matrix(C);

    return 0;
}
